cmake_minimum_required(VERSION 3.24)
project(realmz_music_importer VERSION 1 LANGUAGES CXX)
find_package(libopenmpt CONFIG REQUIRED)
find_package(SndFile CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(realmz-music-importer main.cpp audio_import.cpp)
target_compile_features(realmz-music-importer PRIVATE cxx_std_17)
target_link_libraries(realmz-music-importer PRIVATE libopenmpt::libopenmpt SndFile::sndfile nlohmann_json::nlohmann_json)
if(NOT MUSIC_SOURCE_ID)
  message(FATAL_ERROR "Build through tools/build_music_importer.ps1 to supply the source identity")
endif()
target_compile_definitions(realmz-music-importer PRIVATE MUSIC_SOURCE_ID="${MUSIC_SOURCE_ID}")
if(MSVC)
  target_compile_options(realmz-music-importer PRIVATE /W4 /utf-8)
else()
  target_compile_options(realmz-music-importer PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS realmz-music-importer RUNTIME DESTINATION .)
